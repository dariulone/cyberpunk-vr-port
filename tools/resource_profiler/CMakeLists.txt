cmake_minimum_required(VERSION 3.24)
project(CyberpunkVRProfiler VERSION 1.0.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build with MSVC for Windows x64")
endif()

add_executable(CyberpunkVRProfiler main.cpp)
target_compile_features(CyberpunkVRProfiler PRIVATE cxx_std_20)
target_compile_definitions(CyberpunkVRProfiler PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(CyberpunkVRProfiler PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_link_libraries(CyberpunkVRProfiler PRIVATE pdh dxgi advapi32)
set_property(TARGET CyberpunkVRProfiler PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(CTest)
if(BUILD_TESTING)
    add_executable(profiler_tests tests.cpp)
    target_compile_features(profiler_tests PRIVATE cxx_std_20)
    target_compile_definitions(profiler_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(profiler_tests PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    set_property(TARGET profiler_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    add_test(NAME profiler_metrics COMMAND profiler_tests)
    add_executable(profiler_probe integration_probe.cpp)
    target_compile_features(profiler_probe PRIVATE cxx_std_20)
    target_compile_definitions(profiler_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_compile_options(profiler_probe PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    target_link_libraries(profiler_probe PRIVATE d3d11)
    set_property(TARGET profiler_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
