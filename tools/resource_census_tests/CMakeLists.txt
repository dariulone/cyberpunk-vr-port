cmake_minimum_required(VERSION 3.24)
project(ResourceCensusTests LANGUAGES CXX)
enable_testing()
add_executable(resource_census_tests main.cpp ../../src/Render/ResourceLifetimeCensus.cpp)
target_include_directories(resource_census_tests PRIVATE ../../include)
target_compile_features(resource_census_tests PRIVATE cxx_std_20)
target_compile_definitions(resource_census_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(resource_census_tests PRIVATE d3d12 dxgi)
foreach(case lifetime overflow concurrent duplicate_concurrent)
    add_test(NAME resource_census_${case} COMMAND resource_census_tests ${case})
    set_tests_properties(resource_census_${case} PROPERTIES TIMEOUT 30)
endforeach()
