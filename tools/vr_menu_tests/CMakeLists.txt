cmake_minimum_required(VERSION 3.20)
project(vr_menu_native_tests LANGUAGES CXX ASM_MASM)
add_executable(world_touch_abi world_touch_abi.cpp world_touch_abi.asm ../../src/Hooks/WorldTouchTriangleThunk.asm)
enable_testing()
add_test(NAME world_touch_abi COMMAND world_touch_abi)
add_executable(swipe_tests swipe_tests.cpp)
target_compile_features(swipe_tests PRIVATE cxx_std_20)
target_include_directories(swipe_tests PRIVATE ../../include)
if(MSVC)
    target_compile_options(swipe_tests PRIVATE /W4 /permissive-)
endif()
foreach(case free_swipe no_grip slow slanted cooldown ladder_climb ladder_top_exit ladder_jump_off regrip_after_exit wheel_let_go tracking_lost)
    add_test(NAME swipe_${case} COMMAND swipe_tests ${case})
endforeach()
# The same cases against the Rust detector (rust/crates/cvr-rig/src/menu_swipe.rs through cvr_rust.lib; MenuSwipe.hpp's
# Rust-build class).
option(CVR_VR_MENU_TESTS_RUST "Also run the swipe cases against the Rust detector" ON)
if(CVR_VR_MENU_TESTS_RUST)
    set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
    include(${root}/cmake/RustStaticlib.cmake)
    add_executable(swipe_tests_rust swipe_tests.cpp)
    target_compile_features(swipe_tests_rust PRIVATE cxx_std_20)
    target_include_directories(swipe_tests_rust PRIVATE ../../include)
    target_compile_definitions(swipe_tests_rust PRIVATE CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
    cvr_link_rust(swipe_tests_rust)
    foreach(case free_swipe no_grip slow slanted cooldown ladder_climb ladder_top_exit ladder_jump_off regrip_after_exit wheel_let_go tracking_lost)
        add_test(NAME swipe_${case}_rust COMMAND swipe_tests_rust ${case})
    endforeach()
endif()
