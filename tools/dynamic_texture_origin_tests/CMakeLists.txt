cmake_minimum_required(VERSION 3.24)
project(DynamicTextureOriginTests LANGUAGES CXX)
enable_testing()
add_executable(dynamic_texture_origin_tests main.cpp)
add_executable(dynamic_texture_origin_forwarding forwarding.cpp)
target_include_directories(dynamic_texture_origin_tests PRIVATE ../../include)
target_include_directories(dynamic_texture_origin_forwarding PRIVATE ../../include ../../externals/minhook/include)
target_compile_features(dynamic_texture_origin_tests PRIVATE cxx_std_20)
target_compile_features(dynamic_texture_origin_forwarding PRIVATE cxx_std_20)
target_compile_definitions(dynamic_texture_origin_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_definitions(dynamic_texture_origin_forwarding PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(dynamic_texture_origin_tests PRIVATE /W4 /permissive-)
  target_compile_options(dynamic_texture_origin_forwarding PRIVATE /W4 /permissive-)
endif()
foreach(case overwrite parallel)
  add_test(NAME texture_origin_${case} COMMAND dynamic_texture_origin_tests ${case})
endforeach()
add_test(NAME texture_origin_forwarding COMMAND dynamic_texture_origin_forwarding)
