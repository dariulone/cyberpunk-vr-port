cmake_minimum_required(VERSION 3.24)
project(DebugGateTests LANGUAGES CXX)
enable_testing()
add_executable(debug_gate_tests main.cpp ../../src/Utils/DebugGate.cpp)
target_include_directories(debug_gate_tests PRIVATE ../../include)
target_compile_features(debug_gate_tests PRIVATE cxx_std_20)
add_test(NAME launcher_debug_gate COMMAND debug_gate_tests)

# The gate applied by Rust (rust/crates/cvr-host through cvr_rust.lib) over the same table.
option(CVR_DEBUG_GATE_TESTS_RUST "Also run the test against the Rust gate" ON)
if(CVR_DEBUG_GATE_TESTS_RUST)
    include(../../cmake/RustStaticlib.cmake)
    add_executable(debug_gate_tests_rust main.cpp ../../src/Utils/DebugGate.cpp)
    target_include_directories(debug_gate_tests_rust PRIVATE ../../include)
    target_compile_features(debug_gate_tests_rust PRIVATE cxx_std_20)
    target_compile_definitions(debug_gate_tests_rust PRIVATE CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
    cvr_link_rust(debug_gate_tests_rust)
    add_test(NAME launcher_debug_gate_rust COMMAND debug_gate_tests_rust)
endif()
