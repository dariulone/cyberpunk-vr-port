cmake_minimum_required(VERSION 3.24)
project(MirrorResourceTests LANGUAGES CXX)
enable_testing()
add_executable(mirror_resource_tests main.cpp)
target_include_directories(mirror_resource_tests PRIVATE ../../include)
target_compile_features(mirror_resource_tests PRIVATE cxx_std_20)
target_compile_definitions(mirror_resource_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(mirror_resource_tests PRIVATE /W4 /permissive-)
endif()
foreach(case rebuilds parallel reentrant_release capture_rebuilds capture_parallel capture_reentrant)
    add_test(NAME mirror_resource_${case} COMMAND mirror_resource_tests ${case})
    set_tests_properties(mirror_resource_${case} PROPERTIES TIMEOUT 30)
endforeach()
