cmake_minimum_required(VERSION 3.24)
project(ScannerInputTests LANGUAGES CXX)
enable_testing()
# Compile the production hand-stop function with small stubs, not a copied implementation.
set(hand_source "${CMAKE_CURRENT_SOURCE_DIR}/../../src/Anim/CharacterRig.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${hand_source}")
file(READ "${hand_source}" hand_code)
string(FIND "${hand_code}" "void VRIK_ApplyHandStop(" hand_start)
string(FIND "${hand_code}" "void VRIK_BuildHandTarget(" hand_end)
if(hand_start LESS 0 OR hand_end LESS hand_start)
    message(FATAL_ERROR "Production hand-stop function markers changed")
endif()
math(EXPR hand_length "${hand_end}-${hand_start}")
string(SUBSTRING "${hand_code}" ${hand_start} ${hand_length} hand_stop)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/scanner_hand_stop.inc" "${hand_stop}")
add_executable(scanner_input_tests main.cpp)
target_include_directories(scanner_input_tests PRIVATE ../../include "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(scanner_input_tests PRIVATE cxx_std_20)
add_test(NAME scanner_input COMMAND scanner_input_tests)
