cmake_minimum_required(VERSION 3.20)
project(FixedVrsTests LANGUAGES CXX)
enable_testing()
add_executable(vrs_settings settings.cpp ../../src/Render/FixedVrsSettings.cpp)
target_include_directories(vrs_settings PRIVATE ../../include)
target_compile_features(vrs_settings PRIVATE cxx_std_20)
target_compile_definitions(vrs_settings PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME vrs_settings COMMAND vrs_settings)
add_executable(vrs_gpu gpu.cpp ../../src/Render/FixedVrs.cpp ../../src/Render/FixedVrsSettings.cpp ../../src/Render/ViewInstancedPipeline.cpp ../../src/Render/CommandResources.cpp)
# FixedVrs.cpp asks engine stereo whether the rate map applies (Stereo/EngineStereo.hpp, which includes OpenXR).
set(CVR_OPENXR_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../build/_deps/openxr-src/include")
target_include_directories(vrs_gpu PRIVATE ../../include ${CVR_OPENXR_INCLUDE})
target_compile_features(vrs_gpu PRIVATE cxx_std_20)
target_compile_definitions(vrs_gpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(vrs_gpu PRIVATE d3d12 dxgi d3dcompiler)
add_test(NAME vrs_gpu COMMAND vrs_gpu)

# The policy against Rust (rust/crates/cvr-render through cvr_rust.lib), as the DLL calls it.
option(CVR_VRS_TESTS_RUST "Also run the settings test against the Rust policy" ON)
if(CVR_VRS_TESTS_RUST)
    include(../../cmake/RustStaticlib.cmake)
    add_executable(vrs_settings_rust settings.cpp ../rust_test_support/plugin_stubs.cpp)
    target_include_directories(vrs_settings_rust PRIVATE ../../include)
    target_compile_features(vrs_settings_rust PRIVATE cxx_std_20)
    target_compile_definitions(vrs_settings_rust PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
    cvr_link_rust(vrs_settings_rust)
    add_test(NAME vrs_settings_rust COMMAND vrs_settings_rust)
    # The GPU half against Rust (rust/crates/cvr-render/src/fixed_vrs.rs): FixedVrs.cpp keeps the exports, the
    # engine-stereo query and the pipeline-stream reader.
    add_executable(vrs_gpu_rust gpu.cpp ../../src/Render/FixedVrs.cpp ../../src/Render/ViewInstancedPipeline.cpp ../rust_test_support/plugin_stubs.cpp)
    target_include_directories(vrs_gpu_rust PRIVATE ../../include ${CVR_OPENXR_INCLUDE})
    target_compile_features(vrs_gpu_rust PRIVATE cxx_std_20)
    target_compile_definitions(vrs_gpu_rust PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
    target_link_libraries(vrs_gpu_rust PRIVATE d3d12 dxgi d3dcompiler)
    cvr_link_rust(vrs_gpu_rust)
    add_test(NAME vrs_gpu_rust COMMAND vrs_gpu_rust)
endif()
