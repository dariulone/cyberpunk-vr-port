cmake_minimum_required(VERSION 3.20)
project(FixedVrsTests LANGUAGES CXX)
enable_testing()
set(OPENXR_INCLUDE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../build/cyberfinger-deps/OpenXR-SDK-release-1.0.34/include" CACHE PATH "OpenXR headers")
add_executable(vrs_settings settings.cpp ../../src/Render/FixedVrsSettings.cpp)
target_include_directories(vrs_settings PRIVATE ../../include ${OPENXR_INCLUDE_DIR})
target_compile_features(vrs_settings PRIVATE cxx_std_20)
target_compile_definitions(vrs_settings PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME vrs_settings COMMAND vrs_settings)
add_executable(vrs_gpu gpu.cpp ../../src/Render/FixedVrs.cpp ../../src/Render/FixedVrsSettings.cpp ../../src/Render/ViewInstancedPipeline.cpp ../../src/Render/CommandResources.cpp)
target_include_directories(vrs_gpu PRIVATE ../../include ${OPENXR_INCLUDE_DIR})
target_compile_features(vrs_gpu PRIVATE cxx_std_20)
target_compile_definitions(vrs_gpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(vrs_gpu PRIVATE d3d12 dxgi d3dcompiler)
add_test(NAME vrs_gpu COMMAND vrs_gpu)
add_executable(eye_tracking eye_tracking.cpp ../../src/Runtimes/EyeTracking.cpp)
target_include_directories(eye_tracking PRIVATE ../../include ${OPENXR_INCLUDE_DIR})
target_compile_features(eye_tracking PRIVATE cxx_std_20)
target_compile_definitions(eye_tracking PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
add_test(NAME eye_tracking COMMAND eye_tracking)
set(OPENXR_LOADER_LIBRARY "" CACHE FILEPATH "Optional static OpenXR loader for the manual runtime capability query")
if(OPENXR_LOADER_LIBRARY)
    add_executable(eye_runtime_probe runtime_probe.cpp)
    target_include_directories(eye_runtime_probe PRIVATE ${OPENXR_INCLUDE_DIR})
    target_compile_features(eye_runtime_probe PRIVATE cxx_std_20)
    target_link_libraries(eye_runtime_probe PRIVATE ${OPENXR_LOADER_LIBRARY} advapi32 ole32)
endif()
