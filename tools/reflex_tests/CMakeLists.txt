cmake_minimum_required(VERSION 3.24)
project(ReflexOptionsTests LANGUAGES CXX)
enable_testing()
add_executable(reflex_options main.cpp)
target_include_directories(reflex_options PRIVATE ../../include)
target_compile_features(reflex_options PRIVATE cxx_std_20)
add_test(NAME reflex_options_preserve_native_fields COMMAND reflex_options)

# The same test against the Rust policy (rust/crates/cvr-rig/src/reflex.rs through cvr_rust.lib).
option(CVR_REFLEX_TESTS_RUST "Also run the test against the Rust policy" ON)
if(CVR_REFLEX_TESTS_RUST)
    include(../../cmake/RustStaticlib.cmake)
    add_executable(reflex_options_rust main.cpp)
    target_include_directories(reflex_options_rust PRIVATE ../../include)
    target_compile_features(reflex_options_rust PRIVATE cxx_std_20)
    target_compile_definitions(reflex_options_rust PRIVATE CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
    cvr_link_rust(reflex_options_rust)
    add_test(NAME reflex_options_preserve_native_fields_rust COMMAND reflex_options_rust)
endif()
