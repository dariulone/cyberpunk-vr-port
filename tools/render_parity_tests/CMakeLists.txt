# The same cases against the Rust rules (rust/crates/cvr-render through cvr_rust.lib: fog_history.rs and
# cascade_sampling.rs through render_parity_abi.rs), with the C++ callbacks the library can reach stubbed.
option(CVR_RENDER_PARITY_TESTS_RUST "Also run the tests against the Rust rules" ON)
if(CVR_RENDER_PARITY_TESTS_RUST)
    include(../../cmake/RustStaticlib.cmake)
    add_library(render_parity_rust_stubs STATIC ../single_pass_tests/rust_stubs.cpp)
    set_property(TARGET cvr_rust APPEND PROPERTY INTERFACE_LINK_LIBRARIES render_parity_rust_stubs)
    foreach(cvr_parity_case main fog sky cascade)
        add_executable(render_parity_${cvr_parity_case}_rust ${cvr_parity_case}.cpp)
        target_include_directories(render_parity_${cvr_parity_case}_rust PRIVATE ../../include)
        target_compile_features(render_parity_${cvr_parity_case}_rust PRIVATE cxx_std_20)
        target_compile_definitions(render_parity_${cvr_parity_case}_rust PRIVATE CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
        cvr_link_rust(render_parity_${cvr_parity_case}_rust)
        add_test(NAME render_parity_${cvr_parity_case}_rust COMMAND render_parity_${cvr_parity_case}_rust)
    endforeach()
endif()
