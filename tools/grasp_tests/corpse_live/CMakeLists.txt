cmake_minimum_required(VERSION 3.24)
project(CorpseLiveProbe LANGUAGES CXX)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(JSON_INCLUDE "${REPO}/build/_deps/nlohmann_json-src/single_include" CACHE PATH "Existing nlohmann JSON headers")
if(NOT EXISTS "${JSON_INCLUDE}/nlohmann/json.hpp")
  message(FATAL_ERROR "Set JSON_INCLUDE to the project's existing nlohmann include directory")
endif()
add_executable(corpse_probe probe.cpp)
target_include_directories(corpse_probe PRIVATE "${JSON_INCLUDE}")
target_compile_features(corpse_probe PRIVATE cxx_std_20)
enable_testing()
add_test(NAME corpse_probe_math COMMAND corpse_probe self-test)
