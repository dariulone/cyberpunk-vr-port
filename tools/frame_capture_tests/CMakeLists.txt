cmake_minimum_required(VERSION 3.20)
project(frame_capture_tests LANGUAGES CXX)
enable_testing()
set(OPENXR_INCLUDE "" CACHE PATH "Production OpenXR headers")
add_executable(frame_capture_tests main.cpp ../../src/Framegen/FrameCapture.cpp)
target_compile_features(frame_capture_tests PRIVATE cxx_std_20)
target_include_directories(frame_capture_tests PRIVATE ../../include "${OPENXR_INCLUDE}")
target_compile_definitions(frame_capture_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(frame_capture_tests PRIVATE d3d12 dxgi)
add_test(NAME frame_capture COMMAND frame_capture_tests)
