cmake_minimum_required(VERSION 3.24)
project(LadderClimbingTests LANGUAGES CXX)
enable_testing()
add_executable(ladder_tests main.cpp)
target_compile_features(ladder_tests PRIVATE cxx_std_20)
target_include_directories(ladder_tests PRIVATE ../../include)
if(MSVC)
    target_compile_options(ladder_tests PRIVATE /W4 /permissive-)
endif()
foreach(case geometry top_exit measured_top wrist_lock rung_wrist auto_finish finger_settings directions two_hands release discontinuities closed_loop state)
    add_test(NAME ladder_${case} COMMAND ladder_tests ${case})
endforeach()
