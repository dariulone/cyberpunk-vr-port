cmake_minimum_required(VERSION 3.24)
project(face_tests LANGUAGES CXX)
enable_testing()
add_executable(face_input main.cpp)
target_compile_features(face_input PRIVATE cxx_std_20)
target_include_directories(face_input PRIVATE ../../include)
add_test(NAME face_input COMMAND face_input)
add_executable(face_mapping mapping.cpp)
target_compile_features(face_mapping PRIVATE cxx_std_20)
target_include_directories(face_mapping PRIVATE ../../include)
add_test(NAME face_mapping COMMAND face_mapping)
if(WIN32)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/socket")
    add_executable(face_socket socket.cpp ../../src/Anim/FaceTracking.cpp)
    target_compile_features(face_socket PRIVATE cxx_std_20)
    target_include_directories(face_socket PRIVATE ../../include)
    target_compile_definitions(face_socket PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX
        FACE_TEST_DIR="${CMAKE_CURRENT_BINARY_DIR}/socket")
    target_link_libraries(face_socket PRIVATE ws2_32)
    add_test(NAME face_socket COMMAND face_socket)
endif()
