cmake_minimum_required(VERSION 3.24)
project(HudSubmitTests LANGUAGES CXX)
enable_testing()
set(HUD_OPENXR_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../build-local/_deps/openxr-src/include" CACHE PATH "OpenXR headers directory")
if(NOT EXISTS "${HUD_OPENXR_INCLUDE}/openxr/openxr.h")
    message(FATAL_ERROR "Set HUD_OPENXR_INCLUDE to the OpenXR headers used by the production build")
endif()
add_executable(hud_submit_tests main.cpp ../../src/Runtimes/HudQuad.cpp ../../src/Runtimes/PresentationQueue.cpp)
target_include_directories(hud_submit_tests PRIVATE ../../include "${HUD_OPENXR_INCLUDE}")
target_compile_features(hud_submit_tests PRIVATE cxx_std_20)
target_compile_definitions(hud_submit_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D12)
target_link_libraries(hud_submit_tests PRIVATE d3d12 dxgi)
add_test(NAME hud_submit_ring COMMAND hud_submit_tests)
set_tests_properties(hud_submit_ring PROPERTIES TIMEOUT 15)
