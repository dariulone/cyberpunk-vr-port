cmake_minimum_required(VERSION 3.24)
project(DescriptorCacheTests LANGUAGES CXX)
enable_testing()
add_executable(descriptor_cache_tests main.cpp)
target_compile_features(descriptor_cache_tests PRIVATE cxx_std_20)
target_include_directories(descriptor_cache_tests PRIVATE ../../include)
foreach(case startup_churn active replacement resize concurrent all_pinned image_identity reserve)
    add_test(NAME descriptor_${case} COMMAND descriptor_cache_tests ${case})
endforeach()

# The same cases against the Rust cache (rust/crates/cvr-render/src/descriptor_cache.rs through cvr_rust.lib), with the
# C++ callbacks the library can reach stubbed.
option(CVR_DESCRIPTOR_CACHE_TESTS_RUST "Also run the tests against the Rust cache" ON)
if(CVR_DESCRIPTOR_CACHE_TESTS_RUST)
    include(../../cmake/RustStaticlib.cmake)
    add_library(descriptor_cache_rust_stubs STATIC ../single_pass_tests/rust_stubs.cpp)
    set_property(TARGET cvr_rust APPEND PROPERTY INTERFACE_LINK_LIBRARIES descriptor_cache_rust_stubs)
    add_executable(descriptor_cache_tests_rust main.cpp)
    target_compile_features(descriptor_cache_tests_rust PRIVATE cxx_std_20)
    target_include_directories(descriptor_cache_tests_rust PRIVATE ../../include)
    target_compile_definitions(descriptor_cache_tests_rust PRIVATE CVR_RUST_VRIK=1 CVR_RUST_TEST_FACADES=1)
    cvr_link_rust(descriptor_cache_tests_rust)
    foreach(case startup_churn active replacement resize concurrent all_pinned image_identity reserve)
        add_test(NAME descriptor_${case}_rust COMMAND descriptor_cache_tests_rust ${case})
    endforeach()
endif()
