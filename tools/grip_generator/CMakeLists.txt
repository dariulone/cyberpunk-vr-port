cmake_minimum_required(VERSION 3.24)
project(GripGenerator LANGUAGES CXX)
enable_testing()
find_path(CVR_JSON_INCLUDE nlohmann/json.hpp
    HINTS ${CMAKE_CURRENT_LIST_DIR}/../../build/patchday/sdk25-build/_deps/nlohmann_json-src/single_include
    REQUIRED)
add_executable(grip_generator main.cpp)
target_include_directories(grip_generator PRIVATE ${CVR_JSON_INCLUDE})
target_compile_features(grip_generator PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(grip_generator PRIVATE /W4 /EHsc)
endif()
add_executable(grip_profile_tests profile_tests.cpp)
target_include_directories(grip_profile_tests PRIVATE ${CVR_JSON_INCLUDE})
target_compile_features(grip_profile_tests PRIVATE cxx_std_20)
add_test(NAME grip_mesh_profiles COMMAND grip_profile_tests)
add_executable(grasp_part_tests ../grasp_tests/part_tests.cpp)
target_include_directories(grasp_part_tests PRIVATE ${CVR_JSON_INCLUDE} ../grasp_tests/stub)
target_compile_features(grasp_part_tests PRIVATE cxx_std_20)
target_compile_definitions(grasp_part_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
add_test(NAME grasp_part_selection COMMAND grasp_part_tests)
add_executable(grip_pose_baker pose_baker.cpp)
target_include_directories(grip_pose_baker PRIVATE ${CVR_JSON_INCLUDE} ../grasp_tests/stub)
target_compile_features(grip_pose_baker PRIVATE cxx_std_20)
target_compile_definitions(grip_pose_baker PRIVATE _CRT_SECURE_NO_WARNINGS)
add_executable(grip_sector_refs sector_refs.cpp)
target_include_directories(grip_sector_refs PRIVATE ${CVR_JSON_INCLUDE})
target_compile_features(grip_sector_refs PRIVATE cxx_std_20)
add_executable(grip_pose_batch pose_batch.cpp)
target_include_directories(grip_pose_batch PRIVATE ${CVR_JSON_INCLUDE} ../grasp_tests/stub)
target_compile_features(grip_pose_batch PRIVATE cxx_std_20)
target_compile_definitions(grip_pose_batch PRIVATE _CRT_SECURE_NO_WARNINGS)
if(MSVC)
    target_compile_options(grip_pose_batch PRIVATE /O2 /EHsc /bigobj)
endif()
